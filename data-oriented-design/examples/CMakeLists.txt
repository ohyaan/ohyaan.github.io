cmake_minimum_required(VERSION 3.16)
project(ecs_design_lab LANGUAGES CXX)
include(CTest)
option(DESIGN_LAB_BUILD_EXTENSIONS "Build optional identity/storage experiments" OFF)
function(design_target target source)
  add_executable(${target} ${source})
  target_compile_features(${target} PRIVATE cxx_std_17)
  set_target_properties(${target} PROPERTIES CXX_EXTENSIONS OFF)
  if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
    target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic -Werror)
  endif()
endfunction()
design_target(design_demo demo.cpp)
design_target(design_tests tests.cpp)
if(BUILD_TESTING)
  foreach(group baseline acceleration alternatives identity ecs deferred debug maintenance)
    add_test(NAME design_${group} COMMAND design_tests ${group})
  endforeach()
endif()
if(DESIGN_LAB_BUILD_EXTENSIONS)
  add_subdirectory(../advanced/examples optional)
endif()
